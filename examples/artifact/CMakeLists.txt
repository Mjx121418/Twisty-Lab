cmake_minimum_required(VERSION 3.25)
project(artifact_consumer LANGUAGES CXX)
find_package(TwistyKernel 0.7 CONFIG REQUIRED)
add_executable(artifact_consumer main.cpp)
target_link_libraries(artifact_consumer PRIVATE Twisty::artifact_kernel)
add_executable(artifact_validate validate.cpp)
target_link_libraries(artifact_validate PRIVATE Twisty::artifact_kernel)
option(TWISTY_BUILD_ARTIFACT_EXPORTER "Build the optional authoring example" OFF)
if(TWISTY_BUILD_ARTIFACT_EXPORTER)
  add_executable(artifact_export export.cpp)
  target_link_libraries(artifact_export PRIVATE Twisty::authoring_geometry)
endif()
